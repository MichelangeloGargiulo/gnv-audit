// ── STATO IN MEMORIA ─────────────────────────────────────────

var slots = [];

var cur = 0;

var dept = 'retail';

var open = {};


function today(){ return new Date().toISOString().split('T')[0]; }

function uid(){ return 'n'+Date.now(); }

function slot(){ return slots[cur] || slots[0]; }


function tryLS(fn){ try{ return fn(); }catch(e){ return null; } }


function loadStorage(){

  var raw = tryLS(function(){ return localStorage.getItem('gnv4'); });

  if(raw){ try{ var p=JSON.parse(raw); slots=p.slots; cur=p.cur||0; return; }catch(e){} }

  slots = [{id:uid(), meta:{date:today(),nave:'',verificatore:''}, scores:{}, notes:{}}];

  cur = 0;

}


function saveStorage(){

  tryLS(function(){ localStorage.setItem('gnv4', JSON.stringify({slots:slots,cur:cur})); });

}


// ── META ─────────────────────────────────────────────────────

function syncMeta(){

  slot().meta.date = document.getElementById('mdata').value;

  slot().meta.nave = document.getElementById('mnave').value;

  slot().meta.verificatore = document.getElementById('mver').value;

}

function loadMeta(){

  var m = slot().meta;

  document.getElementById('mdata').value = m.date||today();

  document.getElementById('mnave').value = m.nave||'';

  document.getElementById('mver').value = m.verificatore||'';

}

function onMeta(){ syncMeta(); saveStorage(); renderNaveBar(); }


// ── NAVE BAR ─────────────────────────────────────────────────

function renderNaveBar(){

  var bar = document.getElementById('nave-bar');

  var html = '';

  for(var i=0;i<slots.length;i++){

    var s = slots[i];

    var label = (s.meta.nave||'Nave '+(i+1));

    var d = s.meta.date ? ' '+s.meta.date.split('-').slice(1).reverse().join('/') : '';

    var active = i===cur?' active':'';

    html += '<div class="nave-slot'+active+'" onclick="switchSlot('+i+')">'+(label+d)+'<button class="nave-del" onclick="event.stopPropagation();delSlot('+i+')">✕</button></div>';

  }

  html += '<button class="nave-new" onclick="openModal()">+ Nuova nave</button>';

  bar.innerHTML = html;

  document.getElementById('hsub').textContent = (slot().meta.nave||'Audit') + ' — dati in memoria (esporta per salvare)';

}


function switchSlot(i){ syncMeta(); cur=i; loadMeta(); renderNaveBar(); renderDept(dept); updateSummary(); }


function delSlot(i){

  if(slots.length<=1){ alert('Devi avere almeno un audit.'); return; }

  var nome = slots[i].meta.nave||'questo audit';

  if(!confirm('Eliminare l\'audit di '+nome+'?')) return;

  slots.splice(i,1);

  if(cur>=slots.length) cur=slots.length-1;

  saveStorage(); loadMeta(); renderNaveBar(); renderDept(dept); updateSummary();

}

// ── SCORE ────────────────────────────────────────────────────

function k(dId,sn,idx){ return dId+'_'+sn+'_'+idx; }


function setScore(key,score){

  // toggle: click same value again to deselect

  if(slot().scores[key]===score){ delete slot().scores[key]; score=undefined; }

  else { slot().scores[key]=score; }

  for(var s=0;s<=5;s++){

    var b=document.getElementById('b_'+key+'_'+s);

    if(b) b.className='sbt'+(score===s?' s'+s:'');

  }

  var bna=document.getElementById('bna_'+key);

  if(bna) bna.className='sbt sna'+(score==='na'?' active-na':'');

  updSecScore(key);

  updateSummary();

  saveStorage();

}


function setNA(key){

  if(slot().scores[key]==='na'){ delete slot().scores[key]; }

  else { slot().scores[key]='na'; }

  var score=slot().scores[key];

  for(var s=0;s<=5;s++){

    var b=document.getElementById('b_'+key+'_'+s);

    if(b) b.className='sbt';

  }

  var bna=document.getElementById('bna_'+key);

  if(bna) bna.className='sbt sna'+(score==='na'?' active-na':'');

  updSecScore(key);

  updateSummary();

  saveStorage();

}


function updSecScore(key){

  for(var di=0;di<DEPTS.length;di++){

    var d=DEPTS[di];

    for(var si=0;si<d.secs.length;si++){

      var sec=d.secs[si];

      if(key.indexOf(d.id+'_'+sec.n+'_')===0){

        var r=calcSec(d.id,sec.n,sec.items);

        var el=document.getElementById('ss_'+d.id+'_'+sec.n);

        if(el){

          if(r.scored===0){el.textContent='—';el.className='sec-score';return;}

          var pct=Math.round((r.tot/r.max)*100);

          el.textContent=r.tot+'/'+r.max+' ('+pct+'%)';

          el.className='sec-score '+(pct>=80?'g':pct>=60?'o':'r');

        }

      }

    }

  }

}


function calcSec(dId,sn,items){

  var sc=slot().scores, tot=0,max=0,scored=0;

  for(var i=0;i<items.length;i++){

    var v=sc[k(dId,sn,i)];

    if(v==='na') continue; // escludi N/A dal calcolo

    max+=5;

    if(v!==undefined){tot+=v;scored++;}

  }

  return {tot:tot,max:max,scored:scored};

}


function calcDept(dId){

  var d=DEPTS.filter(function(x){return x.id===dId;})[0];

  var tot=0,max=0;

  for(var si=0;si<d.secs.length;si++){

    var sec=d.secs[si];

    for(var i=0;i<sec.items.length;i++){

      var v=slot().scores[k(dId,sec.n,i)];

      if(v==='na') continue;

      max+=5; tot+=(v||0);

    }

  }

  return {tot:tot,max:max};

}

function crcTable(){

  var t=[];

  for(var n=0;n<256;n++){ var c=n; for(var k=0;k<8;k++) c=(c&1)?(0xedb88320^(c>>>1)):(c>>>1); t[n]=c>>>0; }

  return t;

}

var CRC_TABLE=null;

function crc32(bytes){

  if(!CRC_TABLE) CRC_TABLE=crcTable();

  var c=0xffffffff;

  for(var i=0;i<bytes.length;i++) c=CRC_TABLE[(c^bytes[i])&255]^(c>>>8);

  return (c^0xffffffff)>>>0;

}

function u16(a,n){ a.push(n&255,(n>>>8)&255); }

function u32(a,n){ a.push(n&255,(n>>>8)&255,(n>>>16)&255,(n>>>24)&255); }

function concatBytes(parts){

  var out=new Uint8Array(len),off=0;

  for(i=0;i<parts.length;i++){ out.set(parts[i],off); off+=parts[i].length; }

  return out;

}

function gnvCellXml(cell,row,col){

  if(!cell) return '';

  var r=excelCol(col)+row;

  var s=(cell.s!==undefined && cell.s!==null)?' s="'+cell.s+'"':'';

  var v=cell.v;

  if(v===undefined || v===null || v==='') return '<c r="'+r+'"'+s+'/>';

  if(typeof v==='number' && isFinite(v)) return '<c r="'+r+'"'+s+'><v>'+v+'</v></c>';

  return '<c r="'+r+'"'+s+' t="inlineStr"><is><t>'+escXml(v)+'</t></is></c>';

}

function makeGnvWorksheetXml(cells,widths,rowHeights,merges,maxRows,maxCols){

  var xml='<?xml version="1.0" encoding="UTF-8" standalone="yes"?>';

  xml+='<worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships">';

  xml+='<sheetViews><sheetView workbookViewId="0" zoomScale="50" zoomScaleNormal="50"/></sheetViews>';

  xml+='<cols>';

  for(var i=0;i<widths.length;i++) xml+='<col min="'+(i+1)+'" max="'+(i+1)+'" width="'+widths[i]+'" customWidth="1"/>';

  xml+='</cols><sheetData>';

  for(var r=1;r<=maxRows;r++){

    var ht=rowHeights[r-1]||15;

    xml+='<row r="'+r+'" ht="'+ht+'" customHeight="1">';

    var row=cells[r-1]||[];

    for(var c=1;c<=maxCols;c++) if(row[c-1]) xml+=gnvCellXml(row[c-1],r,c);

    xml+='</row>';

  }

  xml+='</sheetData>';

  if(merges.length){

    xml+='<mergeCells count="'+merges.length+'">';

    for(var m=0;m<merges.length;m++) xml+='<mergeCell ref="'+merges[m]+'"/>';

    xml+='</mergeCells>';

  }

  xml+='<pageMargins left="0.25" right="0.25" top="0.25" bottom="0.25" header="0.1" footer="0.1"/>';

  xml+='<pageSetup paperSize="9" orientation="landscape" fitToWidth="1" fitToHeight="0"/>';

  xml+='</worksheet>';

  return xml;

}

  var maxRows=blockRows*strideR-4;

  for(var c=0;c<maxCols;c++){

    var pos=c%strideC;

    widths[c]=pos===0?3.3:pos===1?31.5:pos===2?4.2:pos===3?2.0:pos===4?8.0:3.0;

  }

  function setCell(r,c,v,style){ if(!cells[r-1]) cells[r-1]=[]; cells[r-1][c-1]={v:v,s:style}; }

  for(var bi=0;bi<blocks.length;bi++){

    var br=Math.floor(bi/blocksPerRow), bc=bi%blocksPerRow;

    var r0=br*strideR+1, c0=bc*strideC+1;

    var b=blocks[bi], sec=b.sec;

    setCell(r0,c0+1,b.dept,1); setCell(r0,c0+2,'',1);

    merges.push(excelCol(c0+1)+r0+':'+excelCol(c0+2)+r0);

    setCell(r0+1,c0,sec.n,2);

  }

  var sheetXml=makeGnvWorksheetXml(cells,widths,rowHeights,merges,maxRows,maxCols);

  var nave=m.nave||'audit';

  downloadOrShare(blob,fname);

}
